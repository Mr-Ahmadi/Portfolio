\newcommand{\tstack}[1]{%
    \def\tfirst{1}%
    \foreach \x in {#1} {%
        \ifnum\tfirst=1
            {\small\color{accent}\texttt{\x}}%
            \gdef\tfirst{0}%
        \else
            \tdot{\small\color{accent}\texttt{\x}}%
        \fi
    }%
}

    \begin{tabularx}{\linewidth}{@{}Xr@{}}
        \textbf{\color{darkgray}#2}%

% Secondary line under a header (location / details); wraps safely.
% #1 = left indent, #2 = text.
\newcommand{\cvsublineat}[2]{%
    \setlength{\cvwidth}{\linewidth}%
    \par\nobreak\vspace{-\sublinelift}\noindent
    \hspace*{#1}%
    \parbox[t]{\dimexpr\cvwidth-#1\relax}{\small\color{lightgray}#2}%
    \par\nobreak\vspace{\sublinedrop}%
}
\newcommand{\cvsubline}[1]{\cvsublineat{\cvindent}{#1}}

% Title > subtitle / date, plus a flush-left location line.
\newcommand{\cventry}[4]{%
    \cvhead{#1}{#2}{#3}%
    \cvsublineat{0pt}{\faMapMarker*\;\,#4}%
}
